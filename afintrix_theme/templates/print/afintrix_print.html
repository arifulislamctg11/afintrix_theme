{# Shared print building blocks — Phase 6. Every Afintrix print format is assembled from these macros, so the six documents a client actually sends look like one product and a change to the header is made once. The styling lives in the Afintrix print style, which applies to every format on the site, so the formats themselves carry no CSS. #} {%- macro head(title, number, meta) -%}
{{ title }}
{{ number }}
{%- for label, value in meta -%} {%- if value -%}
{{ label }} {{ value }}
{%- endif -%} {%- endfor -%}
{%- endmacro -%} {%- macro parties(blocks) -%}
{%- for label, name, lines in blocks -%}
{{ label }}
{{ name or "" }}
{{ lines or "" }}
{%- endfor -%}
{%- endmacro -%} {#- columns: a list of (heading, width, align) -#} {%- macro table_open(columns) -%} {%- for heading, width, align in columns -%} {%- endfor -%} {%- endmacro -%} {%- macro table_close() -%}
#{{ heading }}
{%- endmacro -%} {%- macro item_cell(name, description) -%}
{{ name or "" }}
{%- if description and description != name -%}
{{ description }}
{%- endif -%} {%- endmacro -%} {#- rows: a list of (label, value, is_total) -#} {%- macro summary(rows) -%}
{%- for label, value, is_total in rows -%} {%- if value is not none -%} {%- endif -%} {%- endfor -%}
{{ label }} {{ value }}
{%- endmacro -%} {%- macro terms(label, text) -%} {%- if text -%}
{{ label }}
{{ text }}
{%- endif -%} {%- endmacro -%} {%- macro signatures(left, right) -%}
{{ left }}
{{ right }}
{%- endmacro -%}
